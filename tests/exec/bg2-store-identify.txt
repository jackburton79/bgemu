# The store window's Identify page on BG2's layout: the bottom bar's
# second tab switches to it, clicking an unidentified item selects it,
# "Identify" makes it identified and charges the store's identify price.
# Needs: default BG2 starting party/area; store SHOP01 (offers Identify).
# Mouse coordinates: Identify tab (257,433), second item slot (109,198), after his sling,
# Identify button (227,353), Done (500,459).
Run-Action -,82,-,1,RING05,-,0,0
Run-Action -,118,-,1000,-,-,0,0
Run-Action -,150,-,0,SHOP01,-,0,0

Assert-StoreWindow true
