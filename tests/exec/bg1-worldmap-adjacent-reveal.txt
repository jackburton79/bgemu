# Leaving an area through one of its own map edges is supposed to
# reveal that side's AREA_VISIBLE_FROM_ADJACENT-flagged worldmap
# neighbors (real WMP data marks most areas this way instead of relying
# on a script's explicit RevealAreaOnMap() call) - the engine already
# parsed the ARE/WMP area-link table but never consulted it, so leaving
# via a map edge never made anything new selectable on the world map
# (see Core::LoadWorldMap()'s direction parameter, SearchMap::
# EdgeDirection(), WorldMap::_RevealAdjacentAreas()). Needs the default
# starting party/area (no -P/-a override):
# ./bin/BGEmu -p <BG1 path> -D -x tests/exec/bg1-worldmap-adjacent-reveal.txt

# AR2800 starts hidden and is flagged AREA_VISIBLE_FROM_ADJACENT off
# AR2700's own east link (confirmed against real BG1 WORLDMAP.WMP data)
# - neither has a Game::SetAreaMapVisible() override yet at this point.
Run-Action -,110,-,0,AR2700,-,1300,700
Step-Ticks 5

# (5104,1920) is a real "Worldmap exit" search-map cell (value 14) on
# AR2700's own east edge - confirmed by scanning AR2700SR.BMP.
Queue-Action -,JumpToPoint(5104,1920)
Step-Ticks 5
Step-Ticks 5

Assert-AreaMapVisible AR2700,true
Assert-AreaMapVisible AR2800,true

# Being "visible" isn't being reachable: WorldMap's fAreaOffset (which
# slice of the 640x937 bitmap is shown, and - via MouseMoved()'s
# ConvertToArea() - which slice is click-hittable) stayed at its
# RoomBase constructor default of (0,0) forever, since nothing centered
# it on the area the party just left (real IE, and GemRB, do). AR2700/
# AR2800's icons sit around row 470-550 of the bitmap, well outside the
# default (0,0)-anchored 640x398 viewport, so both were undrawn *and*
# unclickable - only an icon that happened to fall in the top ~398 rows
# ever worked. Fixed by WorldMap::_CenterOnArea(). (310,255) is AR2800's
# on-screen icon position once the map is centered on AR2700 (190,495) -
# both confirmed against real BG1 WMP data.
Mouse-Click 310,255
Step-Ticks 5
Assert-Trigger -,true,AreaCheck("AR2800")
