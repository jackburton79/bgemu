#
# Levelling up is the player's move: experience only makes a level possible
# (Actor::CanLevelUp(), the Record screen's Level Up button, control 37, is
# enabled), the level is gained when that button is pressed. It used to happen
# by itself on every XP award, and gave a multiclass character every class's
# level from the whole XP instead of its share.
# Ajantis (paladin, level 2 - 2250 XP) needs 4500 XP for level 3.
Assert-ClassLevel AJANTI,0,2
Assert-CanLevelUp AJANTI,false
Run-Action -,164,-,3000,,,0,0
Step-Ticks 3
Assert-ClassLevel AJANTI,0,2
Assert-CanLevelUp AJANTI,true
Toggle-Record
Invoke-Control GUIREC,2,37
Assert-ClassLevel AJANTI,0,3
Assert-CanLevelUp AJANTI,false
Assert-Trigger AJANTI,true,HPGT(Myself,16)
# The button does nothing when there is no level to gain.
Invoke-Control GUIREC,2,37
Assert-ClassLevel AJANTI,0,3
Toggle-Record

# Coran is a fighter/thief (both level 3) with 9000-10000 XP: his share for each
# class is half of it, below the 5000 the thief needs for level 4.
Create-Creature CORAN,1000,600
Assert-ClassLevel CORAN,0,3
Assert-ClassLevel CORAN,1,3
Assert-CanLevelUp CORAN,false
Run-Action CORAN,259,-,2000,,,0,0
Step-Ticks 3
Assert-CanLevelUp CORAN,true
Level-Up CORAN
# The thief reaches level 4 (5000), the fighter stays at 3 (8000 for level 4).
Assert-ClassLevel CORAN,0,3
Assert-ClassLevel CORAN,1,4
Assert-CanLevelUp CORAN,false

# The spell slots follow the class table: Xan (mage 2, two level 1 slots) at
# level 5 has 4/2/1 slots of levels 1/2/3 (MXSPLWIZ.2DA).
Create-Creature XAN,1200,600
Assert-SpellSlots XAN,1,1,2
Assert-SpellSlots XAN,1,2,0
Run-Action XAN,259,-,30000,,,0,0
Step-Ticks 3
Level-Up XAN
Assert-ClassLevel XAN,0,5
Assert-SpellSlots XAN,1,1,4
Assert-SpellSlots XAN,1,2,2
Assert-SpellSlots XAN,1,3,1
Assert-SpellSlots XAN,1,4,0
