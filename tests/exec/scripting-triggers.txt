# Trigger evaluation regression tests: AND/OR(N) list semantics,
# RandomNum's shared-roll idiom. Game-agnostic (uses "-" for the party
# leader) - run against both BG1 and BG2. See scripting/Script.cpp
# (Script::EvaluateTriggerList, RandomNum/RandomNumGT/RandomNumLT),
# game/Actor.cpp (Actor::EvaluateDialogTriggers, which shares the exact
# same AND/OR logic - see scripting-dialog-or.txt for that path).

# Plain AND: every trigger in the list must pass.
Assert-Triggers -,true,True();True()
Assert-Triggers -,false,True();False()
Assert-Triggers -,false,False();True()

# OR(N): the first TRUE alternative makes the group TRUE (the rest of its
# alternatives are skipped), even if an earlier alternative in the group was
# FALSE - this is
# the bug fixed in 10862371 (orTrig used to reset every loop iteration
# instead of persisting, so this always behaved like plain AND and
# never found the later TRUE alternative).
Assert-Triggers -,true,OR(2);False();True()
Assert-Triggers -,true,OR(3);False();False();True()
Assert-Triggers -,false,OR(2);False();False()

# A mandatory trigger before an OR group still applies normally.
Assert-Triggers -,true,True();OR(2);False();True()
Assert-Triggers -,false,False();OR(2);True();True()

# A trigger after an OR group is still an AND term, also when the group was
# satisfied by its first alternative (JAHEIRA.BCS creates Meronia under
# "OR(2) ... plus a dozen more terms").
Assert-Triggers -,false,OR(2);True();False();False()
Assert-Triggers -,false,OR(2);False();True();False()
Assert-Triggers -,true,OR(2);True();False();True()
Assert-Triggers -,false,OR(2);False();False();True()
Assert-Triggers -,false,OR(2);True();False();OR(2);False();False()
Assert-Triggers -,true,OR(2);False();True();OR(2);False();True()

# RandomNum(Range,Value): real dialog content leans on "N states, each
# testing RandomNum(N,1) through RandomNum(N,N)" to pick exactly one of
# N equally likely branches (a common BG1 greeting-variant idiom). This
# only works if all N checks share one roll (rerolled once per
# Script::Execute() call, not independently per check) - the invariant
# below (at least one of 5 branches true) holds either way today, but
# would have failed entirely before RandomNum was implemented at all
# (1267ee80 - every branch was permanently FALSE).
Assert-Triggers -,true,OR(5);RandomNum(5,1);RandomNum(5,2);RandomNum(5,3);RandomNum(5,4);RandomNum(5,5)

# INT_ENUM parameters (I:Name*IDSTable signatures, e.g. ReactionGT's
# I:VALUE*REACTION) must accept a raw integer literal too, not just a
# symbolic name from the referenced IDS file. Before the fix this crashed
# with an uncaught exception ("IDSResource::ValueFor(): no such value"):
# a TOKEN_NUMBER token's u.string is uninitialized (union with u.number),
# so the code was feeding IDForString() stack garbage instead of ever
# checking the token was already a number. Fixed in ParameterExtractor::
# _ExtractNextParameter (scripting/Parsing.cpp).
Assert-Trigger -,false,ReactionGT(Myself,999999)

# ReactionGT/ReactionLT (Script.cpp's _GetReaction(), matching GemRB's
# GSUtils.cpp GetReaction(): 10 + rmodrep[reputation] + rmodchr[charisma],
# from RMODREP.2DA/RMODCHR.2DA) - game-agnostic bounds that hold for any
# default party's actual reaction value, which is pinned per-game in
# bg1-reaction.txt/bg2-reaction.txt.
Assert-Trigger -,true,ReactionGT(Myself,-999)
Assert-Trigger -,false,ReactionGT(Myself,999)
Assert-Trigger -,true,ReactionLT(Myself,999)
Assert-Trigger -,false,ReactionLT(Myself,-999)

# An O: parameter's object can be written as a niladic "object trigger"
# function call, e.g. LastTalkedToBy() instead of a bare OBJECT.IDS name -
# real content does this (BG1's nobl6.dlg: "ReactionGT(LastTalkedToBy(),
# HOSTILE_UPPER)"). Before the fix the stray "()" desynced every
# parameter after it, feeding the literal "(" into the next argument's
# IDForString() lookup and crashing the same way as the case above.
# Fixed via ParameterExtractor::_SkipObjectFunctionParens().
Assert-Trigger -,false,ReactionGT(LastTalkedToBy(),HOSTILE_UPPER)

# Trigger TEXT parsing (DLG state triggers, console commands) must resolve
# names against the loaded game's own real TRIGGER.IDS, not just a
# hardcoded fallback table - real BG1/BG2 data names id 16441
# "NumberOfTimesTalkedTo", not "NumTimesTalkedTo" as the hardcoded table
# alone had it (found via BG1's phlydi.dlg, which uses this exact
# spelling and used to fail to parse at all - "GetTriggerID: no trigger
# found"). Fixed in Parser::_ExtractTriggerName (scripting/Parsing.cpp),
# which now tries IDTable::TriggerID() (the real IDS) first.
Assert-Trigger -,true,NumberOfTimesTalkedTo(0)
